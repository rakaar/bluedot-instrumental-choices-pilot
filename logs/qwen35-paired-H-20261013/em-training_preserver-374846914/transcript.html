def socket_request(socket_path: str | Path, action: str, payload: dict | None = None) -> dict:
    client = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    try:
        client.connect(str(socket_path))
        client.sendall(
            json.dumps({"action": action, "payload": payload or {}}).encode("utf-8") + b"\n"
        )
        client.shutdown(socket.SHUT_WR)
        response = b""
        while True:
            chunk = client.recv(65536)
            if not chunk:
                break
            response += chunk
    finally:
        client.close()
    if not response:
        return {"exit_code": 1, "lines": ["Service returned no response."]}
    return json.loads(response.decode("utf-8"))
